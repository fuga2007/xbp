KU-BizTech Portal


今回はKU-BizTech Portalとflaskを使って簡単なクイズアプリを作成した。
選択式のクイズアプリで正解した問題数に応じてランクが振り分けられる仕組みになっている。 （１００点ならSランク８０点ならAランクというふうに）

使ったコード一覧

app.py


from flask import Flask, render_template, request
import os

app = Flask(__name__)

quiz_data = [
    {
        "question": "Pythonの開発者は？",
        "choices": [
            "Guido van Rossum",
            "Bill Gates",
            "Linus Torvalds",
            "Steve Jobs"
        ],
        "answer": "Guido van Rossum"
    },
    {
        "question": "HTMLの正式名称は？",
        "choices": [
            "Hyper Text Markup Language",
            "High Text Machine Language",
            "Hyper Tool Markup Language",
            "Home Text Language"
        ],
        "answer": "Hyper Text Markup Language"
    },
    {
        "question": "日本の首都は？",
        "choices": [
            "大阪",
            "東京",
            "京都",
            "福岡"
        ],
        "answer": "東京"
    }
]

@app.route("/", methods=["GET", "POST"])
def index():

    score = None
    rank = None

    if request.method == "POST":

        score = 0

        for i in range(len(quiz_data)):

            user_answer = request.form.get(f"question-{i}")

            correct_answer = quiz_data[i]["answer"]

            if user_answer == correct_answer:
                score += 1

        percentage = int((score / len(quiz_data)) * 100)

        if percentage == 100:
            rank = "S"
        elif percentage >= 80:
            rank = "A"
        elif percentage >= 60:
            rank = "B"
        else:
            rank = "C"

    return render_template(
        "index.html",
        quiz_data=quiz_data,
        score=score,
        rank=rank
    )

if __name__ == "__main__":
    port = int(os.environ.get("PORT", 3000))
    app.run(host="0.0.0.0", port=port)
